    35→    35→class InMemoryCookieJar : CookieJar {
    36→    36→    private val cookiesList = mutableListOf<Cookie>()
    37→    37→
    38→    38→    @Synchronized
    39→    39→    override fun saveFromResponse(url: HttpUrl, cookies: List<Cookie>) {
    40→    40→        for (cookie in cookies) {
    41→    41→            cookiesList.removeAll { it.name.equals(cookie.name, ignoreCase = true) }
    42→    42→            cookiesList.add(cookie)
    43→    43→        }
    44→    44→    }
    45→    45→
    46→    46→    @Synchronized
    47→    47→    override fun loadForRequest(url: HttpUrl): List<Cookie> {
    48→    48→        return cookiesList.toList()
    49→    49→    }
    50→    50→
    51→    51→    @Synchronized
    52→    52→    fun getAllCookiesHeader(): String {
    53→    53→        return cookiesList.joinToString("; ") { "${it.name}=${it.value}" }
    54→    54→    }
    55→    55→}

   872→   872→        val list = mutableListOf<AttendanceHistoryEntity>()
   873→   873→        val oneDay = 86400000L
   874→   874→        val now = System.currentTimeMillis()
   875→   875→        val dateFormat = SimpleDateFormat("dd-MM-yyyy", Locale.US)
   876→   876→

   886→   886→                )

   889→   889→                    list.add(

   900→   900→
   901→   901→            if (subject.totalClasses > 0) {
   902→   902→                var remainingAttended = subject.attendedClasses
   903→   903→                var remainingMissed = subject.totalClasses - subject.attendedClasses
   904→   904→                for (i in 0 until subject.totalClasses) {
