     1→/**
     2→ * AGC ERP (agclms.in) student portal client — with a ban-aware resilience layer.
     3→ *
     4→ * Portal flow (verified against the live portal):
     5→ *  1. GET  https://agclms.in/Elogin/StudentLogin
     6→ *     -> collect .AspNetCore.Antiforgery.* cookie + __RequestVerificationToken from the form
     7→ *  2. POST StudentId / Password / __RequestVerificationToken (same URL, cookies attached)
     8→ *     -> 302 to /DashBoardStudent on success; login form re-rendered on failure
     9→ *  3. GET  /DashBoardStudent -> profile (name, section, dept, incharge) + subject table rows
    10→ *     each row links to /DashBoardStudent/AttendanceReport?SAId=...
    11→ *  4. GET  each AttendanceReport link -> <tbody> rows: <td>dd-MM-yyyy</td>...PRESENT/ABSENT
    12→ *
    13→ * Resilience layer (evidence-based — the portal's IIS firewall dynamically filters
    14→ * requests to /Elogin/StudentLogin from datacenter IPs by User-Agent identity):
    15→ *  - Honest plain client identity first; automatic rotation across identities on
    16→ *    rejection, with sticky memory of whichever identity last succeeded.
    17→ *  - Global request pacing: all portal requests are serialized with a minimum gap and
    18→ *    jitter, so the app never looks like a burst bot to the portal firewall.
    19→ *  - Circuit breaker: any HTTP 403/429 opens a global cooldown (exponential, capped).
    20→ *    During cooldown NO request is sent — hammering would only make things worse.
    21→ *  - Bounded retries with backoff inside the caller's deadline budget.
    22→ *  - Portal-session reuse: callers may pass previously stored authenticated cookies so
    23→ *    a sync skips the login page entirely (1 request instead of 3+).
    24→ *  - Optional egress relay via PORTAL_PROXY_URL (CONNECT tunnel; TLS stays end-to-end
    25→ *    so credentials are never visible to the relay in plaintext).
    26→ *
    27→ * No third-party HTML parser needed — the portal markup is stable Bootstrap/Razor output.
    28→ */
    29→
    30→const PORTAL_ORIGIN = "https://agclms.in";
    31→const LOGIN_URL = `${PORTAL_ORIGIN}/Elogin/StudentLogin`;
    32→const DASHBOARD_URL = `${PORTAL_ORIGIN}/DashBoardStudent`;
    33→
    34→/**
    35→ * Identity candidates, in preference order (EVIDENCE-BASED, 2026-09-15):
    36→ *
    37→ * Root cause of the HTTP 403s: the portal's IIS firewall rejects requests that
    38→ * present full BROWSER User-Agent strings (Chrome/Safari/Gecko patterns) from
    39→ * non-allowed IP ranges, while plain HTTP-client identities pass through and
    40→ * receive the real login page. Verified deterministically from this server:
    41→ *   plain "Mozilla/5.0"       -> 200 + antiforgery token   (repeatedly)
    42→ *   full Chrome UA            -> 403                        (repeatedly, same minute)
    43→ *   plain UA via Node fetch   -> 200 + token
    44→ *   Chrome UA via Node fetch  -> 403
    45→ *
    46→ * So the primary identity is an honest, plain HTTP-client label (no fake browser
    47→ * claim), and the rotation is kept as self-healing insurance: if the college ever
    48→ * flips the rule, retries automatically fall over to the next identity and the
    49→ * last identity that succeeded is remembered and preferred.
    50→ */
    51→const UA_CANDIDATES = ["Mozilla/5.0",
    52→  "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
    53→  "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36",
    54→];

    57→const PROBE_TIMEOUT_MS = 15_000;
    58→
    59→/** Minimum enforced gap between ANY two requests to the portal (global).
    60→ *  Kept human-slow on purpose: the portal firewall also runs a volume limiter
    61→ *  that temporarily 403s an IP after request bursts. */
    62→const MIN_REQUEST_GAP_MS = 1_200;
    63→/** Extra random jitter added to the gap (0..N ms) — avoids metronome patterns. */
    64→const REQUEST_GAP_JITTER_MS = 800;

    66→/** Cooldown opened after a 403/429 (doubles per consecutive rejection, capped).

    68→ *  re-triggers the firewall's volume limiter and extends the block. */

   144→let lastRequestAt = 0;
   145→let cooldownUntil = 0;
   146→let consecutiveBlocks = 0;
   147→let lastGoodUa: string | null = null;
   148→
   149→// Serializes every portal request through one gate, enforcing the minimum gap.
   150→let gateTail: Promise<unknown> = Promise.resolve();
   151→
   152→const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
   153→const jitter = (ms: number) => Math.round(ms * (0.85 + Math.random() * 0.3));
   154→
   155→function scheduleSlot<T>(fn: () => Promise<T>): Promise<T> {
   156→  const run = gateTail.then(fn, fn);
   157→  gateTail = run.then(
   158→    () => undefined,
   159→    () => undefined
   160→  );
   161→  return run as Promise<T>;
   162→}

   164→function notePortalBlocked() {
   165→  consecutiveBlocks += 1;
   166→  const cd = Math.min(COOLDOWN_BASE_MS * 2 ** (consecutiveBlocks - 1), COOLDOWN_MAX_MS);
   167→  cooldownUntil = Math.max(cooldownUntil, Date.now() + cd);
   168→  console.log(
   169→    `[portal] rejected by portal firewall (HTTP 403/429) — global cooldown ${Math.round(cd / 1000)}s (consecutive: ${consecutiveBlocks})`
   170→  );
   171→}
   172→
   173→function notePortalOk() {
   174→  consecutiveBlocks = 0;
   175→  cooldownUntil = 0;
   176→}
   177→
   178→function blockedRetryAfterSec(): number {
   179→  return Math.max(1, Math.ceil((cooldownUntil - Date.now()) / 1000));
   180→}
   181→
   182→/** True while the global circuit breaker is open — callers should skip, not wait. */
   183→export function isPortalCoolingDown(): boolean {
   184→  return Date.now() < cooldownUntil;
   185→}

   187→/**
   188→ * Waits out an active cooldown. Throws PORTAL_BLOCKED if the caller's deadline
   189→ * cannot cover the remaining cooldown plus one request attempt.
   190→ */
   191→async function awaitCooldown(deadlineMs: number): Promise<void> {
   192→  const remaining = cooldownUntil - Date.now();
   193→  if (remaining <= 0) return;
   194→  if (Date.now() + remaining + REQUEST_TIMEOUT_MS + 2_000 > deadlineMs) {
   195→    throw new PortalError(
   196→      "PORTAL_BLOCKED",
   197→      "The college portal is temporarily rate-limiting our server. Please try again in a few minutes — your saved attendance stays available meanwhile.",
   198→      blockedRetryAfterSec()
   199→    );
   200→  }
   201→  await sleep(remaining + 250);
   202→}

   204→/** Picks the User-Agent for attempt #i, preferring the last known-good identity. */
   205→function uaForAttempt(attempt: number): string {
   206→  const order = lastGoodUa
   207→    ? [lastGoodUa, ...UA_CANDIDATES.filter((u) => u !== lastGoodUa)]
   208→    : UA_CANDIDATES;
   209→  return order[attempt % order.length];
   210→}

   236→function collectCookies(jar: CookieJar, res: Response) {
   237→  const raw: string[] =
   238→    typeof (res.headers as unknown as { getSetCookie?: () => string[] }).getSetCookie === "function"
   239→      ? (res.headers as unknown as { getSetCookie: () => string[] }).getSetCookie()
   240→      : [res.headers.get("set-cookie") ?? ""].filter(Boolean);
   241→  for (const line of raw) {
   242→    const [pair] = line.split(";");
   243→    const idx = pair.indexOf("=");
   244→    if (idx > 0) {
   245→      const name = pair.slice(0, idx).trim();
   246→      const value = pair.slice(idx + 1).trim();
   247→      jar.set(name, value);
   248→    }
   249→  }
   250→}
   251→
   252→function cookieHeader(jar: CookieJar): string {
   253→  return Array.from(jar.entries())
   254→    .map(([k, v]) => `${k}=${v}`)
   255→    .join("; ");
   256→}

   258→function decodeEntities(s: string): string {
   259→  return s
   260→    .replace(/&amp;/g, "&")
   261→    .replace(/&lt;/g, "<")
   262→    .replace(/&gt;/g, ">")
   263→    .replace(/&quot;/g, '"')
   264→    .replace(/&#39;|&apos;/g, "'")
   265→    .replace(/&nbsp;/g, " ")
   266→    .replace(/&#(\d+);/g, (_, d) => String.fromCharCode(Number(d)));
   267→}
   268→
   269→function stripTags(html: string): string {
   270→  return decodeEntities(html.replace(/<[^>]*>/g, " ")).replace(/\s+/g, " ").trim();
   271→}

   273→/** Delays (ms) before each attempt for a given purpose. Index 0 = immediate.
   274→ *  Deliberately minimal: the portal's volume limiter lasts minutes, so repeated
   275→ *  quick retries only add fuel. Interactive fails fast (UI serves cached data);
   276→ *  background gets one well-spaced retry. */
   277→function attemptDelays(purpose: PortalPurpose): number[] {
   278→  return purpose === "background" ? [0, 90_000] : [0, 5_000];

   299→async function rawFetch(
   300→  url: string,
   301→  init: RequestInit,
   302→  timeoutMs: number
   303→): Promise<Response> {
   304→  const dispatcher = await relayDispatcher();
   305→  try {
   306→    if (dispatcher) {
   307→      const { fetch: undiciFetch } = await import("undici");
   308→      return (await undiciFetch(url, {
   309→        ...init,
   310→        dispatcher,
   311→        signal: AbortSignal.timeout(timeoutMs),
   312→      } as never)) as unknown as Response;
   313→    }
   314→    return await fetch(url, {
   315→      ...init,
   316→      signal: AbortSignal.timeout(timeoutMs),
   317→      cache: "no-store",
   318→    });
   319→  } catch (e) {
   320→    const msg = e instanceof Error ? e.message : String(e);
   321→    if (/timeout|abort/i.test(msg)) throw new Error("__timeout__");
   322→    throw new Error("__network__");
   323→  }
   324→}

   326→/**
   327→ * Single paced request to the portal (no retries — retries live in portalFetch).
   328→ * Serialized globally with a jittered minimum gap between requests.
   329→ */
   330→async function pacedRequest(
   331→  url: string,
   332→  init: PortalFetchInit,
   333→  cfg: PortalFetchConfig
   334→): Promise<{ res: Response; html: string }> {
   335→  return scheduleSlot(async () => {
   336→    const gap = lastRequestAt + jitter(MIN_REQUEST_GAP_MS + REQUEST_GAP_JITTER_MS / 2) - Date.now();
   337→    if (gap > 0) await sleep(gap);
   338→    lastRequestAt = Date.now();
   339→
   340→    const ua = cfg.__ua ?? uaForAttempt(0);
   341→    const headers: Record<string, string> = {
   342→      "User-Agent": ua,
   343→      Accept: "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8",
   344→      "Accept-Language": "en-US,en;q=0.9",
   345→      ...(init.headers as Record<string, string> | undefined),
   346→    };
   347→    const cookie = cookieHeader(cfg.jar);
   348→    if (cookie) headers.Cookie = cookie;
   349→
   350→    const res = await rawFetch(
   351→      url,
   352→      {
   353→        method: init.method ?? "GET",
   354→        headers,
   355→        body: init.body,
   356→        redirect: init.redirect ?? "follow",
   357→      },
   358→      cfg.timeoutMs ?? REQUEST_TIMEOUT_MS
   359→    );
   360→    collectCookies(cfg.jar, res);
   361→    const html = await res.text();
   362→    return { res, html };
   363→  });
   364→}

   366→/**
   367→ * Resilient portal request: paced, cooldown-aware, with bounded retries,
   368→ * identity rotation on 403/429, and short retries for transient network/5xx.
   369→ */
   370→async function portalFetch(
   371→  url: string,
   372→  init: PortalFetchInit,
   373→  cfg: PortalFetchConfig
   374→): Promise<{ res: Response; html: string }> {
   375→  const resilient = cfg.resilient !== false;
   376→  const delays = resilient ? attemptDelays(cfg.purpose) : [0];
   377→
   378→  let lastError: Error | null = null;
   379→  let lastRes: { res: Response; html: string } | null = null;
   380→
   381→  for (let attempt = 0; attempt < delays.length; attempt++) {
   382→    const requestMs = cfg.timeoutMs ?? REQUEST_TIMEOUT_MS;
   383→    const wait = jitter(delays[attempt]);
   384→    if (wait > 0) {
   385→      // Stop if the next attempt cannot fit in the caller's budget.
   386→      if (Date.now() + wait + requestMs > cfg.deadlineMs) break;
   387→      await sleep(wait);
   388→    }
   389→
   390→    try {
   391→      await awaitCooldown(cfg.deadlineMs);
   392→    } catch (e) {
   393→      // Cooldown cannot fit the budget — surface only if nothing succeeded yet.
   394→      if (lastRes) return lastRes;
   395→      throw e;
   396→    }
   397→
   398→    let out: { res: Response; html: string };
   399→    try {
   400→      out = await pacedRequest(url, { ...init, headers: { ...init.headers } }, { ...cfg, __ua: uaForAttempt(attempt) });
   401→    } catch (e) {
   402→      const kind = e instanceof Error ? e.message : "";
   403→      lastError =
   404→        kind === "__timeout__"
   405→          ? new PortalError("TIMEOUT", "The college portal took too long to respond. Please try again.")
   406→          : new PortalError("NETWORK", "Could not reach the college portal. Check your connection and try again.");
   407→      continue; // transient — try the next attempt
   408→    }
   409→
   410→    const status = out.res.status;
   411→    if (status === 403 || status === 429) {
   412→      notePortalBlocked();
   413→      lastRes = out;
   414→      lastError = new PortalError(
   415→        "PORTAL_BLOCKED",
   416→        "The college portal is temporarily blocking our server. Please try again in a few minutes.",
   417→        blockedRetryAfterSec()
   418→      );
   419→      continue; // rotate identity / wait for cooldown on the next attempt
   420→    }
   421→
   422→    if (!resilient) {
   423→      // Probe mode: hand every non-blocked response back — the caller interprets
   424→      // status + markup (e.g. an unauthenticated dashboard returns 500 BY DESIGN,
   425→      // which signals bad credentials, not an outage).
   426→      if (status < 400) notePortalOk();
   427→      return out;
   428→    }
   429→
   430→    if (status >= 500) {
   431→      lastRes = out;
   432→      lastError = new PortalError(
   433→        "PORTAL_DOWN",
   434→        `The college portal seems to be having trouble right now (HTTP ${status}). Please try again shortly.`
   435→      );
   436→      continue; // transient — retry with short backoff
   437→    }
   438→
   439→    if (status < 400) notePortalOk();
   440→    return out;
   441→  }
   442→
   443→  if (lastRes) {
   444→    // We did get responses — report the strongest error for the final state.
   445→    const status = lastRes.res.status;
   446→    if (status === 403 || status === 429) {
   447→      throw new PortalError(
   448→        "PORTAL_BLOCKED",
   449→        "The college portal is temporarily blocking our server (its firewall is rate-limiting automated access). This usually clears within a few minutes — please try again shortly. Your saved attendance stays available meanwhile.",
   450→        blockedRetryAfterSec()
   451→      );
   452→    }
   453→    if (status >= 500) {
   454→      throw new PortalError(
   455→        "PORTAL_DOWN",
   456→        `The college portal seems to be having trouble right now (HTTP ${status}). Please try again shortly.`
   457→      );
   458→    }
   459→    return lastRes;
   460→  }
   461→  throw lastError ?? new PortalError("NETWORK", "Could not reach the college portal.");
   462→}

   550→function parseSubjectRows(dashboardHtml: string): SubjectRow[] {
   551→  const rows: SubjectRow[] = [];
   552→  const trRe = /<tr[^>]*>([\s\S]*?)<\/tr>/gi;
   553→  let m: RegExpExecArray | null;
   554→
   555→  while ((m = trRe.exec(dashboardHtml)) !== null) {
   556→    const row = m[1];
   557→    if (!/AttendanceReport\?/i.test(row)) continue;
   558→
   559→    const hrefMatch =
   560→      row.match(/href=["']([^"']*AttendanceReport\?[^"']+)["']/i) ||
   561→      row.match(/href=["']([^"']*AttendanceReport[^"']*)["']/i);
   562→    if (!hrefMatch) continue;
   563→    const href = decodeEntities(hrefMatch[1]).trim();
   564→
   565→    // Extract every <td> cell text in order.
   566→    const cells: string[] = [];
   567→    const tdRe = /<td[^>]*>([\s\S]*?)<\/td>/gi;
   568→    let td: RegExpExecArray | null;
   569→    while ((td = tdRe.exec(row)) !== null) {
   570→      const text = stripTags(td[1]);
   571→      if (text) cells.push(text);
   572→    }
   573→
   574→    const saMatch = href.match(/SAId=(\d+)/i);
   575→    const name = cells[0] ?? `Subject ${rows.length + 1}`;
   576→    // Some rows may include extra cells (faculty etc.); second cell is the type when present.
   577→    const type = cells[1] && cells[1].length <= 24 ? cells[1] : "Theory";
   578→
   579→    if (name && name.toLowerCase() !== "subject") {
   580→      rows.push({
   581→        name,
   582→        type,
   583→        href: href.startsWith("http") ? href : `${PORTAL_ORIGIN}${href.startsWith("/") ? "" : "/"}${href}`,
   584→        saId: saMatch ? saMatch[1] : null,
   585→      });
   586→    }
   587→  }
   588→  return rows;
   589→}
   590→
   591→const DATE_RE = /\b(\d{2})[-/](\d{2})[-/](\d{4})\b/;
   592→
   593→function parseAttendanceReport(html: string): { attended: number; total: number; logs: PortalLogEntry[] } {
   594→  let attended = 0;
   595→  let total = 0;
   596→  const logs: PortalLogEntry[] = [];
   597→
   598→  const tbodies: string[] = [];
   599→  const tbodyRe = /<tbody[^>]*>([\s\S]*?)<\/tbody>/gi;
   600→  let tb: RegExpExecArray | null;
   601→  while ((tb = tbodyRe.exec(html)) !== null) tbodies.push(tb[1]);
   602→
   603→  const scope = tbodies.length > 0 ? tbodies.join("\n") : html;
   604→
   605→  const trRe = /<tr[^>]*>([\s\S]*?)<\/tr>/gi;
   606→  let tr: RegExpExecArray | null;
   607→  while ((tr = trRe.exec(scope)) !== null) {
   608→    const row = tr[1];
   609→    const rowText = stripTags(row);
   610→    const dm = rowText.match(DATE_RE);
   611→    if (!dm) continue;
   612→
   613→    let status: "PRESENT" | "ABSENT" | null = null;
   614→    if (/\bPRESENT\b/i.test(rowText)) status = "PRESENT";
   615→    else if (/\bABSENT\b/i.test(rowText)) status = "ABSENT";
   616→    else if (/\bP\b/.test(rowText) && rowText.replace(/[^A-Za-z]/g, "").length <= 1) status = "PRESENT";
   617→    else if (/\bA\b/.test(rowText) && rowText.replace(/[^A-Za-z]/g, "").length <= 1) status = "ABSENT";
   618→    if (!status) continue;
   619→
   620→    total += 1;
   621→    if (status === "PRESENT") attended += 1;
   622→    logs.push({
   623→      subjectCode: "", // filled by caller
   624→      date: `${dm[1]}-${dm[2]}-${dm[3]}`,
   625→      status,
   626→    });
   627→  }
   628→
   629→  return { attended, total, logs };
   630→}

   634→export interface FetchSnapshotOptions {
   635→  /** Previously stored authenticated portal cookies — skips the login page when valid. */
   636→  existingCookies?: PortalCookie[];
   637→  /** Interactive (user waiting) or background (scheduler) — controls retry budget. */
   638→  purpose?: PortalPurpose;
   639→  /** Absolute epoch-ms budget; defaults per purpose. */
   640→  deadlineMs?: number;
   641→}
   642→
   643→function jarFromCookies(cookies: PortalCookie[] | undefined): CookieJar {
   644→  const jar: CookieJar = new Map();
   645→  for (const c of cookies ?? []) {
   646→    if (c && c.name && typeof c.value === "string") jar.set(c.name, c.value);
   647→  }
   648→  return jar;
   649→}

   651→export async function fetchPortalSnapshot(
   652→  studentId: string,
   653→  password: string | null,
   654→  opts: FetchSnapshotOptions = {}
   655→): Promise<PortalSnapshot> {
   656→  const purpose: PortalPurpose = opts.purpose ?? "interactive";
   657→  const deadlineMs =
   658→    opts.deadlineMs ?? Date.now() + (purpose === "background" ? 8 * 60_000 : 70_000);
   659→
   660→  // ---- Attempt 0: reuse a stored authenticated portal session ----------------
   661→  if (opts.existingCookies && opts.existingCookies.length > 0) {
   662→    const jar = jarFromCookies(opts.existingCookies);
   663→    try {
   664→      const probe = await pacedRequest(
   665→        DASHBOARD_URL,
   666→        { headers: { Referer: LOGIN_URL } },
   667→        { jar, purpose, deadlineMs, timeoutMs: PROBE_TIMEOUT_MS, resilient: false }
   668→      );
   669→      if (probe.res.ok && looksAuthenticated(probe.html)) {
   670→        return await harvestFromDashboard(probe.html, jar, { reused: true, purpose, deadlineMs });
   671→      }
   672→    } catch {
   673→      // Probe failures fall through to a fresh login, which reports real errors.
   674→    }
   675→  }
   676→
   677→  // ---- Fresh login flow ------------------------------------------------------
   678→  if (!password) {
   679→    throw new PortalError(
   680→      "UNKNOWN",
   681→      "Saved portal session has expired and no password is stored. Please log in again with your portal password."
   682→    );
   683→  }
   684→
   685→  const jar: CookieJar = new Map();
   686→
   687→  // Step 1: login page (token + antiforgery cookie).
   688→  const loginPage = await portalFetch(
   689→    LOGIN_URL,
   690→    { method: "GET" },
   691→    { jar, purpose, deadlineMs }
   692→  );
   693→  if (loginPage.res.status >= 400) {
   694→    if (loginPage.res.status === 403 || loginPage.res.status === 429) {
   695→      throw new PortalError(
   696→        "PORTAL_BLOCKED",
   697→        "The college portal is temporarily blocking our server (its firewall is rate-limiting automated access). This usually clears within a few minutes — please try again shortly. Your saved attendance stays available meanwhile.",
   698→        blockedRetryAfterSec()
   699→      );
   700→    }
   701→    throw new PortalError(
   702→      "PORTAL_DOWN",
   703→      `College portal rejected the connection (HTTP ${loginPage.res.status}).`
   704→    );
   705→  }
   706→
   707→  const token = extractVerificationToken(loginPage.html);
   708→  if (!token) {
   709→    throw new PortalError("PORTAL_DOWN", "Could not read the portal security token. The portal may be under maintenance.");
   710→  }
   711→
   712→  // Step 2: submit credentials.
   713→  const body = new URLSearchParams({
   714→    StudentId: studentId.trim(),
   715→    Password: password,
   716→    __RequestVerificationToken: token,
   717→  });
   718→
   719→  const post = await portalFetch(
   720→    LOGIN_URL,
   721→    {
   722→      method: "POST",
   723→      headers: {
   724→        "Content-Type": "application/x-www-form-urlencoded",
   725→        Origin: PORTAL_ORIGIN,
   726→        Referer: LOGIN_URL,
   727→      },
   728→      body: body.toString(),
   729→      redirect: "manual",
   730→    },
   731→    { jar, purpose, deadlineMs }
   732→  );
   733→
   734→  // A 3xx after POST means the server accepted the login and is redirecting.
   735→  let dashboardHtml = "";
   736→  if (post.res.status >= 300 && post.res.status < 400) {
   737→    const loc = post.res.headers.get("location");
   738→    const target = loc
   739→      ? loc.startsWith("http")
   740→        ? loc
   741→        : `${PORTAL_ORIGIN}${loc.startsWith("/") ? "" : "/"}${loc}`
   742→      : DASHBOARD_URL;
   743→    const dash = await portalFetch(
   744→      target,
   745→      { method: "GET", headers: { Referer: LOGIN_URL } },
   746→      { jar, purpose, deadlineMs }
   747→    );
   748→    dashboardHtml = dash.html;
   749→  } else {
   750→    dashboardHtml = post.html;
   751→    // The portal may render the dashboard directly with HTTP 200.
   752→    if (!looksAuthenticated(dashboardHtml)) {
   753→      // NOTE: deliberately non-resilient — an unauthenticated /DashBoardStudent
   754→      // returns HTTP 500 by design on this portal, which is the "bad credentials"
   755→      // signal, not an outage. Retrying it would only add load.
   756→      const dash = await portalFetch(
   757→        DASHBOARD_URL,
   758→        { method: "GET", headers: { Referer: LOGIN_URL } },
   759→        { jar, purpose, deadlineMs, resilient: false }
   760→      );
   761→      if (looksAuthenticated(dash.html)) dashboardHtml = dash.html;
   762→    }
   763→  }
   764→
   765→  // Step 3: verify authentication.
   766→  if (isLoginPage(dashboardHtml) || !looksAuthenticated(dashboardHtml)) {
   767→    throw new PortalError(
   768→      "INVALID_CREDENTIALS",
   769→      "Invalid roll number or password. Please check your AGC ERP credentials and try again."
   770→    );
   771→  }
   772→
   773→  return await harvestFromDashboard(dashboardHtml, jar, { reused: false, purpose, deadlineMs });
   774→}

   776→/* --------------------- shared dashboard harvest ---------------------------- */
   777→
   778→async function harvestFromDashboard(
   779→  dashboardHtml: string,
   780→  jar: CookieJar,
   781→  ctx: { reused: boolean; purpose: PortalPurpose; deadlineMs: number }
   782→): Promise<PortalSnapshot> {
   783→  // Parse profile + subject table.
   784→  const profile = parseProfile(dashboardHtml);
   785→  const subjectRows = parseSubjectRows(dashboardHtml);
   786→
   787→  if (subjectRows.length === 0) {
   788→    throw new PortalError(
   789→      "NO_SUBJECTS",
   790→      "Logged in successfully, but no subjects are visible on your portal dashboard yet."
   791→    );
   792→  }
   793→
   794→  // Fetch each subject's attendance report.
   795→  const subjects: PortalSubject[] = [];
   796→  const logs: PortalLogEntry[] = [];
   797→
   798→  // Requests are globally paced by the gate — sequential here keeps order deterministic.
   799→  for (const row of subjectRows) {
   800→    const code = row.saId ? `AGC-${row.saId}` : `AGC-${row.name.replace(/\s+/g, "-").toUpperCase().slice(0, 24)}`;
   801→    let attended = 0;
   802→    let total = 0;
   803→    let reportOk = false;
   804→    if (Date.now() > ctx.deadlineMs + REQUEST_TIMEOUT_MS) {
   805→      // Flow budget exhausted — keep this subject at its last stored values instead
   806→      // of persisting misleading zeros.
   807→      subjects.push({
   808→        subjectCode: code,
   809→        subjectName: row.name,
   810→        subjectType: row.type,
   811→        saId: row.saId,
   812→        attended: 0,
   813→        total: 0,
   814→        percentage: 0,
   815→        reportOk: false,
   816→      });
   817→      continue;
   818→    }
   819→    try {
   820→      const rep = await portalFetch(
   821→        row.href,
   822→        { method: "GET", headers: { Referer: DASHBOARD_URL } },
   823→        { jar, purpose: ctx.purpose, deadlineMs: Math.min(ctx.deadlineMs, Date.now() + 45_000) }
   824→      );
   825→      if (rep.res.ok) {
   826→        const parsed = parseAttendanceReport(rep.html);
   827→        attended = parsed.attended;
   828→        total = parsed.total;
   829→        for (const l of parsed.logs) logs.push({ ...l, subjectCode: code });
   830→        reportOk = true;
   831→      }
   832→    } catch {
   833→      // Single subject report failing should not fail the whole sync.
   834→    }
   835→
   836→    const pct = total > 0 ? (attended / total) * 100 : 0;
   837→    subjects.push({
   838→      subjectCode: code,
   839→      subjectName: row.name,
   840→      subjectType: row.type,
   841→      saId: row.saId,
   842→      attended,
   843→      total,
   844→      percentage: Math.round(pct * 10) / 10,
   845→      reportOk,
   846→    });
   847→  }
   848→
   849→  return {
   850→    profile,
   851→    subjects,
   852→    logs,
   853→    fetchedAt: new Date().toISOString(),
   854→    cookies: Array.from(jar.entries()).map(([name, value]) => ({ name, value })),
   855→    reusedSession: ctx.reused,
   856→  };
   857→}
